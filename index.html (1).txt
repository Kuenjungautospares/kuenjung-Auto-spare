        function renderTable(data) {
            const tbody = document.getElementById('tableBody');
            tbody.innerHTML = '';
            data.forEach((item, index) => {
                if (!item['Parts Name']) return;
                const row = document.createElement('tr');
                row.innerHTML = `
                    <td>${item['No.'] || index + 1}</td>
                    <td><b>${item['Parts Name']}</b></td>
                    <td>${item['Vehicle Type'] || '-'}</td>
                    <td>${item['Quantity'] || '-'}</td>
                    <td>${item['Price (Nu.)'] ? 'Nu. ' + item['Price (Nu.)'] : '-'}</td>
                    <td>${item['Total (Nu.)'] ? 'Nu. ' + item['Total (Nu.)'] : '-'}</td>
                    <td>${item['Remarks'] || '-'}</td>
                    <td><button class="buy-btn" onclick="addToCart(${index})">Add to Cart</button></td>
                `;
                tbody.appendChild(row);
            });
        }


        function filterTable() {
            const query = document.getElementById('searchInput').value.toLowerCase();
            const filtered = inventoryData.filter(item => {
                const name = String(item['Parts Name'] || '').toLowerCase();
                const vehicle = String(item['Vehicle Type'] || '').toLowerCase();
                const remarks = String(item['Remarks'] || '').toLowerCase();
                return name.includes(query) || vehicle.includes(query) || remarks.includes(query);
            });
            renderTable(filtered);
        }


        function addToCart(index) {
            const item = inventoryData[index];
            cart.push(item);
            document.getElementById('cartCount').innerText = cart.length;
            alert(item['Parts Name'] + ' added to cart!');
        }


        function toggleCart() {
            const modal = document.getElementById('cartModal');
            modal.style.display = modal.style.display === 'flex' ? 'none' : 'flex';
            if (modal.style.display === 'flex') {
                renderCartItems();
            }
        }


        function renderCartItems() {
            const list = document.getElementById('cartItemsList');
            if (cart.length === 0) {
                list.innerHTML = '<p>Your cart is empty.</p>';
                document.getElementById('cartTotal').innerText = 'Total: Nu. 0';
                return;
            }
            let html = '';
            let total = 0;
            cart.forEach((item, i) => {
                const price = parseFloat(item['Price (Nu.)'] || item['Total (Nu.)'] || 0);
                total += price;
                html += `<div class="cart-item"><span>{item['Vehicle Type'] || 'General'})</span> <b>Nu. ${price}</b></div>`;
            });
            list.innerHTML = html;
            document.getElementById('cartTotal').innerText = 'Total: Nu. ' + total;
        }


        function sendWhatsAppOrder() {
            if (cart.length === 0) {
                alert('Your cart is empty!');
                return;
            }
            let message = 'Hello Kuenjung Auto Spares, I would like to order:\n';
            let total = 0;
            cart.forEach((item, i) => {
                const price = parseFloat(item['Price (Nu.)'] || item['Total (Nu.)'] || 0);
                total += price;
                message += `${i+1}. {item['Vehicle Type'] || 'General'}] - Nu. ${price}\n`;
            });
            message += `\nTotal Amount: Nu. ${total}`;
            const encoded = encodeURIComponent(message);
            window.open(`https://wa.me/?text=${encoded}`, '_blank');
        }
